CRUD list

A list of people filtered by the start of the surname, with buttons to create, update and delete.

import 'package:thor_ui/thor_ui.dart';

class Person {
  const Person(this.id, this.name, this.surname);

  final int id;
  final String name;
  final String surname;

  factory Person.fromJson(Map<String, Object?> json) => Person(
    json['id']! as int,
    json['name']! as String,
    json['surname']! as String,
  );

  Map<String, Object?> toJson() => {'id': id, 'name': name, 'surname': surname};
}

class PeopleViewModel extends ThorState {
  final people = state(const [
    Person(1, 'Hans', 'Emil'),
    Person(2, 'Max', 'Mustermann'),
  ]);
  final prefix = state('');
  final selected = state(0);
  final name = TextEditingController();
  final surname = TextEditingController();
  var nextId = 3;

  // The people whose surname starts with the filter, once per change.
  late final visible = derived(
    () => [
      for (final p in people.value)
        if (p.surname.toLowerCase().startsWith(prefix.value.toLowerCase())) p,
    ],
  );

  int idOf(Person p) => p.id;
  String label(Person p) => '${p.surname}, ${p.name}';
  bool isSelected(Person p) => p.id == selected.value;
  bool hasSelection() => selected.value != 0;
  List<Person> rows() => visible.value;

  String shownCount() => '${visible.value.length}';
  String totalCount() => '${people.value.length}';

  void setPrefix(String text) => prefix.value = text;

  void select(int id) => batch(() {
    selected.value = id;
    for (final p in people.value) {
      if (p.id == id) {
        name.text = p.name;
        surname.text = p.surname;
      }
    }
  });

  void create() => batch(() {
    people.value = [...people.value, Person(nextId, name.text, surname.text)];
    selected.value = nextId;
    nextId++;
  });

  void update() => people.value = [
    for (final p in people.value)
      if (p.id == selected.value) Person(p.id, name.text, surname.text) else p,
  ];

  void remove() => batch(() {
    people.value = [
      for (final p in people.value)
        if (p.id != selected.value) p,
    ];
    selected.value = 0;
    name.text = '';
    surname.text = '';
  });
}

final peopleState = PeopleViewModel();

class CrudRecipe extends StatelessWidget {
  const CrudRecipe({super.key});

  @override
  Widget build(BuildContext context) {
    return ConstrainedBox(
      constraints: const BoxConstraints(maxWidth: 420),
      child: Column(
        crossAxisAlignment: CrossAxisAlignment.stretch,
        spacing: 12,
        children: [
          TextField(
            onChanged: peopleState.setPrefix,
            decoration: const InputDecoration(labelText: 'Filter prefix'),
          ),
          Row(
            crossAxisAlignment: CrossAxisAlignment.start,
            spacing: 12,
            children: [
              Expanded(
                child: Container(
                  height: 180,
                  decoration: BoxDecoration(
                    border: Border.all(color: ThemeColor.outline),
                    borderRadius: BorderRadius.circular(12),
                  ),
                  child: SingleChildScrollView(
                    child: Watch.list(
                      peopleState.rows,
                      itemKey: peopleState.idOf,
                      container: (rows) => Semantics(
                        role: SemanticsRole.list,
                        label: 'People',
                        child: Column(
                          crossAxisAlignment: CrossAxisAlignment.stretch,
                          children: rows,
                        ),
                      ),
                      each: (row) => Semantics(
                        role: SemanticsRole.listItem,
                        child: row.watch(
                          peopleState.isSelected,
                          builder: (on) => Container(
                            padding: const EdgeInsets.symmetric(
                              horizontal: 8,
                              vertical: 2,
                            ),
                            color: on ? ThemeColor.primary : null,
                            // web: a text child makes it a native <button>.
                            child: InkWell(
                              onTap: peopleState.select.withArg(row.key),
                              child: row.text(
                                peopleState.label,
                                style: TextStyle(
                                  color: on ? ThemeColor.onPrimary : null,
                                ),
                              ),
                            ),
                          ),
                        ),
                      ),
                    ),
                  ),
                ),
              ),
              Expanded(
                child: Column(
                  crossAxisAlignment: CrossAxisAlignment.stretch,
                  spacing: 8,
                  children: [
                    TextField(
                      controller: peopleState.name,
                      decoration: const InputDecoration(labelText: 'Name'),
                    ),
                    TextField(
                      controller: peopleState.surname,
                      decoration: const InputDecoration(labelText: 'Surname'),
                    ),
                  ],
                ),
              ),
            ],
          ),
          Row(
            spacing: 8,
            children: [
              OutlinedButton(
                onPressed: peopleState.create,
                child: const Text('Create'),
              ),
              Watch(
                peopleState.hasSelection,
                builder: (on) => OutlinedButton(
                  onPressed: on ? peopleState.update : null,
                  child: const Text('Update'),
                ),
              ),
              Watch(
                peopleState.hasSelection,
                builder: (on) => OutlinedButton(
                  onPressed: on ? peopleState.remove : null,
                  child: const Text('Delete'),
                ),
              ),
            ],
          ),
          Row(
            spacing: 4,
            children: [
              Watch(peopleState.shownCount),
              const Text('of'),
              Watch(peopleState.totalCount),
            ],
          ),
        ],
      ),
    );
  }
}

The filter is a computed value

people holds every person and visible, a computed value (derived), filters them once per change. Create, update and delete replace the list inside a batch, so the page updates only once.

class Person {
  const Person(this.id, this.name, this.surname);

  final int id;
  final String name;
  final String surname;

  factory Person.fromJson(Map<String, Object?> json) => Person(
    json['id']! as int,
    json['name']! as String,
    json['surname']! as String,
  );

  Map<String, Object?> toJson() => {'id': id, 'name': name, 'surname': surname};
}

class PeopleViewModel extends ThorState {
  final people = state(const [Person(1, 'Hans', 'Emil')]);
  final prefix = state('');
  final selected = state(0);

  late final visible = derived(
    () => [
      for (final p in people.value)
        if (p.surname.toLowerCase().startsWith(prefix.value.toLowerCase())) p,
    ],
  );

  List<Person> rows() => visible.value;
  int idOf(Person p) => p.id;
  String label(Person p) => '${p.surname}, ${p.name}';
  bool isSelected(Person p) => p.id == selected.value;
  bool hasSelection() => selected.value != 0;

  void select(int id) => selected.value = id;
  void remove() => batch(() {
    people.value = [
      for (final p in people.value)
        if (p.id != selected.value) p,
    ];
    selected.value = 0;
  });
}

One row per person

Watch.list renders the rows on the server; in the browser it adds, removes or updates only the row that changed, identified by itemKey. row.watch and row.text update what changes inside each row without rebuilding it.

Watch.list(
  people.rows,
  itemKey: people.idOf,
  each: (row) => row.watch(
    people.isSelected,
    builder: (on) => InkWell(
      onTap: people.select.withArg(row.key),
      child: row.text(people.label),
    ),
  ),
)

Update and Delete read hasSelection with a Watch: with no selection they receive onPressed: null and stay disabled.
